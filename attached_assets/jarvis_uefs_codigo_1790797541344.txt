    // Avaliador matemático restrito: aceita números e operadores, sem eval ou código arbitrário.
    function calculate(raw) {
      let expression = normalize(raw)
        .replace(/^(jarvis[,!?.\s]*)?(calcule|calcular|quanto (e|eh))\s*/i, '')
        .replace(/(\d+(?:[,.]\d+)?)\s*(?:por cento|%)\s*de\s*/g, '($1/100)*')
        .replace(/\b(mais)\b/g, '+').replace(/\b(menos)\b/g, '-')
        .replace(/\b(multiplicado por|vezes|x)\b/g, '*')
        .replace(/\b(dividido por|dividido|sobre)\b/g, '/')
        .replace(/\b(elevado a)\b/g, '^').replace(/\b(raiz de)\b/g, 'sqrt')
        .replace(/,/g, '.').replace(/[×]/g, '*').replace(/[÷]/g, '/').replace(/[−]/g, '-')
        .replace(/[?=]/g, '').trim();
      if (!expression || expression.length > 120 || !/^[\d\s.+*/()^\-]+$/.test(expression)) return null;
      const tokens = expression.match(/\d+(?:\.\d+)?|[()+*/^\-]/g);
      if (!tokens || tokens.join('') !== expression.replace(/\s/g, '')) return null;
      let i = 0;
      function factor() {
        const token = tokens[i++];
        if (token === '-') return -factor();
        if (token === '+') return factor();
        if (token === '(') {
          const value = sum();
          if (tokens[i++] !== ')') throw Error('Parêntese');
          return value;
        }
        if (!token || !/^\d+(?:\.\d+)?$/.test(token)) throw Error('Número');
        return Number(token);
      }
      function power() {
        const base = factor();
        return tokens[i] === '^' ? (i++, Math.pow(base, power())) : base;
      }
      function product() {
        let value = power();
        while (tokens[i] === '*' || tokens[i] === '/') {
          const op = tokens[i++]; const next = power();
          if (op === '/' && next === 0) throw Error('Divisão por zero');
          value = op === '*' ? value * next : value / next;
        }
        return value;
      }
      function sum() {
        let value = product();
        while (tokens[i] === '+' || tokens[i] === '-') {
          const op = tokens[i++]; const next = product(); value = op === '+' ? value + next : value - next;
        }
        return value;
      }
      try {
        const answer = sum();
        if (i !== tokens.length || !Number.isFinite(answer)) return null;
        return new Intl.NumberFormat('pt-BR', { maximumFractionDigits: 8 }).format(answer);
      } catch { return null; }
    }

      }
      if (/\b(que horas|hora atual|horario)\b/.test(command)) {
        reply('Agora são ' + timeNow() + ', horário de Brasília.'); return;
      }
      if (/\b(que dia|data de hoje|qual a data|hoje e que dia)\b/.test(command)) {

        reply('Hoje é ' + date + '.'); return;
      }

        stopTimer(); reply('Temporizador cancelado.'); return;
      }
      if (/\b(temporizador|contagem regressiva|conte)\b/.test(command)) {
        const match = command.match(/(\d+(?:[,.]\d+)?)\s*(segundos?|secs?|s|minutos?|mins?|min|horas?|h)\b/);
        if (!match) { reply('Informe o período. Por exemplo: temporizador de 5 minutos.'); return; }

      }
      if (/\b(minhas notas|listar notas|mostrar notas|anotacoes)\b/.test(command)) {
        reply(notes.length ? 'Você tem ' + notes.length + ' anotações. A lista está no painel.' : 'Você ainda não tem anotações.'); return;
      }
      if (/\b(limpar notas|apagar notas)\b/.test(command)) {
        notes = []; saveNotes(); renderNotes(); reply('Anotações removidas deste navegador.'); return;
      }
      if (/\b(pesquise|pesquisar|busque|buscar|procure)\b/.test(command)) {
        const query = spoken.replace(/^\s*(jarvis\s*[,.:!?]?\s*)?(pesquise|pesquisar|busque|buscar|procure)\s*(por\s*)?/i, '').trim();
        if (!query) { reply('Diga o que deseja pesquisar.'); return; }
        reply('Pesquisa preparada para: ' + query + '. Abra pelo botão abaixo.', { href: 'https://www.google.com/search?q=' + encodeURIComponent(query), label: 'Abrir pesquisa ↗' }); return;
      }
      if (/\b(tela cheia|modo apresentacao)\b/.test(command)) {
        if (!document.fullscreenEnabled) { reply('Este navegador não permite tela cheia aqui. Use F11 para ampliar a janela.'); return; }
        document.documentElement.requestFullscreen().then(() => reply('Modo apresentação ativado.')).catch(() => reply('Toque na página e tente novamente, ou use F11.')); return;
      }
      if (/^(calcule|calcular|quanto e|quanto eh)\b/.test(command) || /^[\d(\-+]/.test(command)) {
        const result = calculate(command);
        reply(result === null ? 'Não entendi a expressão. Exemplo: calcule 25 vezes 4, ou calcule 30% de 200.' : 'O resultado é ' + result + '.'); return;
      }
      reply('Ainda não tenho esse comando. Diga ajuda para ver o que posso fazer nesta versão.');
    }

    function initSpeech() {
      if (!Recognition) {
        els.mic.disabled = true; els.mic.title = 'Reconhecimento de voz indisponível neste navegador';
        setStatus('Voz de entrada indisponível · use o teclado', 'error-status'); return;
      }
      recognition = new Recognition(); recognition.lang = 'pt-BR';
      recognition.interimResults = false; recognition.continuous = false; recognition.maxAlternatives = 1;
      recognition.onstart = () => { listening = true; els.mic.textContent = '◉ Ouvindo...'; setStatus('Ouvindo comando', 'listening-status'); };
      recognition.onresult = (event) => {
        const transcript = event.results[0]?.[0]?.transcript;
        if (transcript) processCommand(transcript);
      };
      recognition.onerror = (event) => {
        const messages = { 'not-allowed': 'Microfone bloqueado. Autorize o acesso nas permissões do navegador.', 'no-speech': 'Não detectei sua fala. Tente novamente.', 'network': 'Reconhecimento indisponível no momento. Use o teclado.', 'audio-capture': 'Nenhum microfone disponível.' };
        setStatus('Falha na escuta', 'error-status');
        addMessage('jarvis', messages[event.error] || 'Não foi possível reconhecer a fala. Use o teclado.');
      };
      recognition.onend = () => { listening = false; els.mic.textContent = '🎙 Iniciar escuta'; if (!els.status.classList.contains('error-status')) setStatus('Aguardando comando'); };
    }
    els.mic.addEventListener('click', () => {
      if (!recognition) return;
      if (listening) { recognition.stop(); return; }
      window.speechSynthesis?.cancel();
      try { recognition.start(); } catch { addMessage('jarvis', 'Aguarde um instante e tente ativar o microfone novamente.'); }
    });
    els.stop.addEventListener('click', () => {
      if (listening) recognition.stop(); window.speechSynthesis?.cancel(); stopTimer();
      setStatus('Aguardando comando'); addMessage('jarvis', 'Voz e temporizador interrompidos.');
    });
    els.voice.addEventListener('click', () => {
      voiceEnabled = !voiceEnabled;
      els.voice.textContent = voiceEnabled ? '🔊 Voz ligada' : '🔇 Voz desligada';
      els.voice.setAttribute('aria-pressed', String(voiceEnabled));
      if (!voiceEnabled) window.speechSynthesis?.cancel();
    });
    els.form.addEventListener('submit', (event) => { event.preventDefault(); const value = els.input.value; els.input.value = ''; processCommand(value); els.input.focus(); });
    document.querySelectorAll('[data-command]').forEach(button => button.addEventListener('click', () => processCommand(button.dataset.command)));
    updateClock(); setInterval(updateClock, 1000); renderNotes(); initSpeech();
    addMessage('jarvis', 'Sistema iniciado. Toque em um comando de exemplo, digite no console ou ative o microfone.');
  })();
